## Equation
Using: 
- $\text{ast}$ 

$$
\text{bhfc} =
\begin{cases} 
    1 & \text{if ast} \leq \text{ulnast and bili} \leq \text{ulnbili}, \\
    2 & \text{if ast} \text{ast} > \text{ulnast or ulnbili} < \text{bili} \leq 1.5 \cdot \text{ulnbili}, \\
    3 & \text{if } 1.5 \cdot \text{ulnbili} < \text{bili} \leq 3 \cdot \text{ulnbili}, \\
    4 & \text{if bili} > 3 \cdot \text{ulnbili}, \\
    -999 & \text{otherwise.}
\end{cases}
$$


## Code

`dplyr::near()` is used to catch any floating point issues at category boundaries.

```r
 edge_cases <- FALSE
if (
    !any(is.na(bili)) & 
    !any(is.na(ulnbili)) & 
    any(dplyr::near(bili, 1.5 * ulnbili))) {
        edge_cases <- TRUE
}
if (
    !any(is.na(bili)) & 
    !any(is.na(ulnbili)) & 
    any(dplyr::near(bili, 3 * ulnbili))) {
        edge_cases <- TRUE
}

if (!edge_cases) {
    bhfc <- dplyr::case_when(
        ast <= ulnast & bili <= ulnbili                             ~ 1,
        ast > ulnast | dplyr::between(bili, ulnbili, 1.5 * ulnbili) ~ 2,
        dplyr::between(bili, 1.5 * ulnbili, 3 * ulnbili)            ~ 3,
        bili > 3 * ulnbili                                          ~ 4,
        .default = -999
    )

    } else {
    bhfc <- dplyr::case_when(
        # bili is near 1.5 * ulnbili or 3 * ulnbili so it's either 2, 3
        ast > ulnast | dplyr::near(bili, 1.5 * ulnbili) ~ 2,
        dplyr::near(bili, 3 * ulnbili)                  ~ 3,
        .default = -999
    )
}
```
